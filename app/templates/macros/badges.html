{# macros/badges.html — centralised badge macros for MedCover. Design rules enforced here so templates never need to think about them: - bg-warning/bg-info always paired with text-black (CSS-variable aware) instead of text-dark (fixed dark color, not dark-mode aware). - bg-body-secondary used for neutral/muted backgrounds (dark-mode safe). - Never use bg-light or bg-white in app templates. #} {# ── Generic semantic badges ──────────────────────────────────────────────── #} {% macro badge_success(text, extra_classes='') %} {{ text }} {% endmacro %} {% macro badge_danger(text, extra_classes='') %} {{ text }} {% endmacro %} {% macro badge_warning(text, extra_classes='') %} {{ text }} {% endmacro %} {% macro badge_secondary(text, extra_classes='') %} {{ text }} {% endmacro %} {% macro badge_info(text, extra_classes='') %} {{ text }} {% endmacro %} {# ── Event / ME status badge ──────────────────────────────────────────────── #} {# status_colors is a dict mapping status value strings to Bootstrap colour names, e.g. {'Zveřejněno': 'success', 'Koncept': 'secondary', ...}. Pass the same dict that routes already compute and inject into the template. #} {% macro badge_status(status_value, status_colors, extra_classes='') %} {{ status_value }} {% endmacro %} {# ── Debriefing event-note status badge ───────────────────────────────────── #} {% macro badge_event_note_status(status) %} {{ 'Bez poznámek' if status in [0, 1] else ('Popsáno' if status == 2 else 'Probrat') }} {% endmacro %} {# ── Event staffing badges (mandatory + optional) ────────────────────────── #} {# Two-badge display for an event's obsazení: - mandatory badge: red until all mandatory spots are filled, green once full; hidden when the event has no mandatory spots. - optional badge: yellow until all optional spots are filled, green once full; hidden when the event has no optional spots. Pass verbose=true on the event detail page for the full Czech sentence; the default compact form is used on the events list. #} {# Optional `counts` dict lets a caller precompute {mandatory_total, mandatory_filled, optional_total, optional_filled} once per row and skip five per-macro-call passes over event.spots. When omitted, falls back to the Event properties. #} {% macro obsazeni_badges(event, verbose=false, counts=None) %} {%- if counts -%} {%- set mf = counts.mandatory_filled -%} {%- set mt = counts.mandatory_total -%} {%- set of = counts.optional_filled -%} {%- set ot = counts.optional_total -%} {%- else -%} {%- set mf = event.mandatory_filled_spots -%} {%- set mt = event.mandatory_total_spots -%} {%- set of = event.optional_filled_spots -%} {%- set ot = event.optional_total_spots -%} {%- endif -%} {%- set mand_cls = 'bg-success' if mt > 0 and mf >= mt else 'bg-danger' -%} {%- set opt_cls = 'bg-success' if ot > 0 and of >= ot else 'bg-warning text-black' -%} {%- if verbose -%} {%- if mt > 0 %}{{ mf }}/{{ mt }} nutných pozic obsazeno{% endif -%} {%- if ot > 0 %} {{ of }}/{{ ot }} volitelných pozic obsazeno{% endif -%} {%- else -%} {%- if mt > 0 %}{{ mf }}/{{ mt }}{% endif -%} {%- if ot > 0 %} {{ of }}/{{ ot }} vol.{% endif -%} {%- endif -%} {% endmacro %}
